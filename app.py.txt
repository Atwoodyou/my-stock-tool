import streamlit as st
import yfinance as yf
import twstock
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
from datetime import datetime, timedelta

# 恢復為預設白底圖表，符合你的 PS 設計圖
plt.style.use('default')
plt.rcParams['font.sans-serif'] = ['DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

st.set_page_config(page_title="AI 飆股五線譜", layout="wide", page_icon="📈")

# 頂部：切分左右兩欄 (左邊戰情室輸入，右邊 AI 解讀)
top_left, top_right = st.columns([1, 1.5])

with top_left:
    st.markdown("### 📱 個人股票戰情室")
    st.write("") # 增加微小間距
    input_col1, input_col2 = st.columns(2)
    with input_col1:
        stock_input = st.text_input("股票代號", "^TWII", label_visibility="collapsed")
    with input_col2:
        mode_input = st.selectbox("分析週期", ['3.5年 (長線)', '1年 (波段)', '半年 (飆股 - 極短線)'], label_visibility="collapsed")
    go_btn = st.button("開始分析", type="primary", use_container_width=True)

if go_btn or stock_input:
    stock_id = stock_input.strip().upper()
    if not stock_id:
        st.warning("請輸入代號")
        st.stop()

    stock_name = stock_id
    ticker = stock_id
    if stock_id.isdigit():
        ticker = f"{stock_id}.TW"
        if stock_id in twstock.codes:
            stock_name = twstock.codes[stock_id].name
    else:
        try:
            t = yf.Ticker(ticker)
            stock_name = t.info.get('shortName', stock_id)
        except: pass

    if mode_input == '3.5年 (長線)':
        lookback_days = 850
        channel_window = 100
    elif mode_input == '1年 (波段)':
        lookback_days = 250
        channel_window = 60
    else:
        lookback_days = 120
        channel_window = 20

    end_date = datetime.now()
    start_date = end_date - timedelta(days=lookback_days * 2)
    
    try:
        df = yf.download(ticker, start=start_date, end=end_date, progress=False)
        if df.empty and stock_id.isdigit():
            ticker = f"{stock_id}.TWO"
            df = yf.download(ticker, start=start_date, end=end_date, progress=False)
        
        if df.empty:
            st.error(f"❌ 找不到代號 {stock_id}")
            st.stop()

        if isinstance(df.columns, pd.MultiIndex):
            df_close = df.xs('Close', level=0, axis=1).iloc[:, 0]
            df_high = df.xs('High', level=0, axis=1).iloc[:, 0]
            df_low = df.xs('Low', level=0, axis=1).iloc[:, 0]
        else:
            df_close = df['Close']
            df_high = df['High']
            df_low = df['Low']

        if len(df_close) < lookback_days: lookback_days = len(df_close)
        df_calc = pd.DataFrame({'Close': df_close, 'High': df_high, 'Low': df_low}).tail(lookback_days).copy()
        
        y = df_calc['Close'].values
        x = np.arange(len(y))
        slope, intercept, _, _, _ = stats.linregress(x, y)
        trend_line = slope * x + intercept
        std_dev = np.std(y - trend_line)

        lohas_upper = df_calc['High'].rolling(window=channel_window).max()
        lohas_lower = df_calc['Low'].rolling(window=channel_window).min()
        lohas_ma = df_calc['Close'].rolling(window=channel_window).mean()

        curr = y[-1]
        p2sd = (trend_line + 2*std_dev)[-1]
        p1sd = (trend_line + 1*std_dev)[-1]
        m1sd = (trend_line - 1*std_dev)[-1]
        m2sd = (trend_line - 2*std_dev)[-1]

        with top_right:
            st.markdown("#### 💡 AI 投資顧問解讀")
            if curr > p2sd:
                st.error("🚨 **極度危險 (強烈減碼訊號)**\n\n目前股價已突破 +2SD，價格極度昂貴。")
            elif curr > p1sd:
                st.warning("⚠️ **相對高檔 (減碼訊號)**\n\n目前股價位於 +1SD ~ +2SD 之間，價格偏貴。若持有部位可考慮適度調節，空手者不建議此時進場。")
            elif curr < m2sd:
                st.success("💰 **極度便宜 (強烈買進訊號)**\n\n目前股價跌破 -2SD，價格極度便宜。")
            elif curr < m1sd:
                st.info("🟢 **相對低檔 (買進訊號)**\n\n目前股價位於 -1SD ~ -2SD 之間，價格偏便宜。")
            else:
                st.info("⚖️ **價格合理 (觀望或定期定額)**\n\n目前股價位於 -1SD ~ +1SD 之間，評價合理。")
            
            trend_status = "多頭趨勢" if slope > 0 else "空頭趨勢"
            trend_desc = "股價長期重心向上移動，回檔找買點勝率較高。" if slope > 0 else "股價長期重心向下移動，反彈找賣點勝率較高。"
            st.markdown(f"📊 趨勢背景： **{trend_status}** (斜率 {slope:.2f}) - {trend_desc}")

        st.write("") # 增加區塊間距

        # 下方圖表：左右並排
        chart_col1, chart_col2 = st.columns(2)

        with chart_col1:
            st.markdown(f"**{stock_name} - 五線譜位階**")
            fig1, ax1 = plt.subplots(figsize=(8, 4.5))
            # 維持你原本習慣的顏色設定
            ax1.plot(df_calc.index, trend_line + 2*std_dev, 'r--', label='+2SD')
            ax1.plot(df_calc.index, trend_line + 1*std_dev, color='salmon', linestyle='--', alpha=0.5)
            ax1.plot(df_calc.index, trend_line, 'b', label='TL')
            ax1.plot(df_calc.index, trend_line - 1*std_dev, color='lightgreen', linestyle='--', alpha=0.5)
            ax1.plot(df_calc.index, trend_line - 2*std_dev, 'g--', label='-2SD')
            ax1.plot(df_calc.index, y, 'k', label='Price')
            ax1.legend(loc='upper left')
            ax1.grid(True, alpha=0.3)
            # 強制設定為白底，避免被網頁黑底影響
            fig1.patch.set_facecolor('white')
            ax1.set_facecolor('white')
            st.pyplot(fig1)

        with chart_col2:
            st.markdown(f"**動能通道 ({channel_window}日)**")
            fig2, ax2 = plt.subplots(figsize=(8, 4.5))
            ax2.fill_between(df_calc.index, lohas_upper, lohas_lower, color='cyan', alpha=0.1)
            ax2.plot(df_calc.index, lohas_upper, color='teal')
            ax2.plot(df_calc.index, lohas_lower, color='teal')
            ax2.plot(df_calc.index, lohas_ma, color='orange', linestyle='--')
            ax2.plot(df_calc.index, y, 'k')
            ax2.grid(True, alpha=0.3)
            fig2.patch.set_facecolor('white')
            ax2.set_facecolor('white')
            st.pyplot(fig2)

    except Exception as e:
        st.error(f"分析失敗: {e}")
